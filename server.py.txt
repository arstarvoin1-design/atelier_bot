def get_admin_ids() -> list[int]:
    conn = get_db()
    rows = conn.execute("SELECT tg_id FROM admins").fetchall()
    conn.close()
    return [r["tg_id"] for r in rows]


def is_admin_tg(tg_id: int) -> bool:
    return tg_id in get_admin_ids()


def upsert_user(tg_id: int, username: str = "", full_name: str = "", photo_url: str = ""):
    conn = get_db()
    conn.execute(
        """INSERT INTO users (tg_id, username, full_name, photo_url, created_at)
           VALUES (?, ?, ?, ?, ?)
           ON CONFLICT(tg_id) DO UPDATE SET
             username=COALESCE(NULLIF(excluded.username, ''), users.username),
             full_name=COALESCE(NULLIF(excluded.full_name, ''), users.full_name),
             photo_url=COALESCE(NULLIF(excluded.photo_url, ''), users.photo_url)""",
        (tg_id, username, full_name, photo_url, time.time()),
    )
    conn.commit()
    conn.close()


def serialize_product(row: sqlite3.Row, conn: sqlite3.Connection) -> dict:
    pid = row["id"]
    rev_stat = conn.execute(
        "SELECT AVG(rating) as avg_r, COUNT(*) as cnt FROM reviews WHERE product_id = ?", (pid,)
    ).fetchone()
    avg_rating = round(float(rev_stat["avg_r"]), 1) if rev_stat and rev_stat["avg_r"] is not None else 0.0
    review_count = int(rev_stat["cnt"]) if rev_stat else 0

    return {
        "id": pid,
        "group_slug": row["group_slug"] or "other",
        "name": row["name"],
        "price": row["price"],
        "fabric": row["fabric"] or "",
        "description": row["description"] or "",
        "photos": json.loads(row["photos_json"] or "[]"),
        "stock": json.loads(row["stock_json"] or "{}"),
        "size_chart": json.loads(row["size_chart_json"] or "[]"),
        "rating_avg": avg_rating,
        "review_count": review_count,
    }


def format_price(num: int) -> str:
    return f"{int(num):,}".replace(",", " ") + " ₽"


def verify_telegram_webapp_init_data(init_data: str) -> Optional[dict]:
    if not init_data or not BOT_TOKEN:
        return None
    try:
        parsed = dict(urllib.parse.parse_qsl(init_data, strict_parsing=True))
        received_hash = parsed.pop("hash", "")
        data_check_string = "\n".join(f"{k}={v}" for k, v in sorted(parsed.items()))
        secret_key = hmac.new(b"WebAppData", BOT_TOKEN.encode("utf-8"), hashlib.sha256).digest()
        calculated_hash = hmac.new(secret_key, data_check_string.encode("utf-8"), hashlib.sha256).hexdigest()
        if hmac.compare_digest(calculated_hash, received_hash):
            return json.loads(parsed.get("user", "{}"))
    except Exception:
        return None
    return None


def verify_telegram_login_widget(data: dict) -> bool:
    if not BOT_TOKEN or "hash" not in data:
        return False
    check_data = {k: str(v) for k, v in data.items() if k != "hash" and v is not None and v != ""}
    received_hash = str(data["hash"])
    data_check_string = "\n".join(f"{k}={v}" for k, v in sorted(check_data.items()))
    secret_key = hashlib.sha256(BOT_TOKEN.encode("utf-8")).digest()
    calculated_hash = hmac.new(secret_key, data_check_string.encode("utf-8"), hashlib.sha256).hexdigest()
    return hmac.compare_digest(calculated_hash, received_hash)

            "full_name": full_name,
            "username": username,

@app.get("/api/orders/{tg_id}")

        "SELECT * FROM orders WHERE user_tg_id = ? ORDER BY created_at DESC", (tg_id,)
    ).fetchall()
    conn.close()

    active, archive = [], []
    for r in rows:
        item = dict(r)
        item["items"] = json.loads(item.pop("items_json") or "[]")
        if item["is_archived"] or item["status"] in ("Доставлено", "Отменен"):
            archive.append(item)
        else:
            active.append(item)
    return {"active": active, "archive": archive}

def can_user_review_product(conn: sqlite3.Connection, tg_id: int, product_id: int) -> bool:
    if not tg_id:
        return False
    rows = conn.execute(
        "SELECT items_json FROM orders WHERE user_tg_id = ? AND status = 'Доставлено'", (tg_id,)
    ).fetchall()
    for r in rows:
        items = json.loads(r["items_json"] or "[]")
        if any(int(it.get("product_id") or 0) == product_id for it in items):
            return True
    return False


@app.get("/api/products/{product_id}/reviews")
async def api_get_reviews(product_id: int, tg_id: int = 0):
    conn = get_db()
    rows = conn.execute(
        "SELECT * FROM reviews WHERE product_id = ? ORDER BY created_at DESC", (product_id,)
    ).fetchall()
    can_review = can_user_review_product(conn, tg_id, product_id)
    already = False
    if tg_id:
        ex = conn.execute(
            "SELECT id FROM reviews WHERE product_id = ? AND user_tg_id = ?", (product_id, tg_id)
        ).fetchone()
        already = bool(ex)
    conn.close()

    return {
        "reviews": [dict(r) for r in rows],
        "can_review": can_review and not already,
        "already_reviewed": already,
    }

async def create_product_from_tg(message: types.Message, raw_text: str, file_ids: list[str]):
    payload = raw_text.partition(" ")[2].strip()
    parts = [p.strip() for p in payload.split("|")]
    if len(parts) < 3:
        await message.answer(
            "➕ Формат: <code>/add группа | Название | Цена | S:5, M:5, L:5 | Описание</code>\n"
            "Например: <code>/add tee | Футболка ATELIER Oversize | 3200 | S:4, M:6, L:3 | Плотный хлопок</code>",
            parse_mode="HTML",
        )
        return

    group_slug = parts[0].lower()
    name = parts[1]
    price = int("".join(c for c in parts[2] if c.isdigit()) or 0)
    stock = {}
    if len(parts) > 3 and parts[3]:
        for entry in parts[3].split(","):
            if ":" in entry:
                sz, cnt = entry.split(":", 1)
                stock[sz.strip()] = int("".join(c for c in cnt if c.isdigit()) or 0)
            elif entry.strip():
                stock[entry.strip()] = 5
    if not stock:
        stock = {"S": 5, "M": 5, "L": 5, "XL": 5}
    desc = parts[4] if len(parts) > 4 else ""

    photos = [await save_tg_photo_locally(fid, i) for i, fid in enumerate(file_ids)]
    default_chart = [{"size": sz, "chest": "—", "shoulders": "—", "length": "—"} for sz in stock.keys()]

    conn = get_db()
    conn.execute(
        """INSERT INTO products (group_slug, name, price, fabric, description, photos_json, stock_json, size_chart_json, created_at)
           VALUES (?, ?, ?, '100% хлопок', ?, ?, ?, ?, ?)""",
        (group_slug, name, price, desc, json.dumps(photos), json.dumps(stock), json.dumps(default_chart), time.time()),
    )
    conn.commit()
    conn.close()
    await message.answer(f"✅ Товар <b>«{html.escape(name)}»</b> добавлен в группу <code>{html.escape(group_slug)}</code>!", parse_mode="HTML")

@dp.callback_query(F.data.startswith("ord:"))
async def cb_admin_order_status(callback: types.CallbackQuery):
    if not is_admin_tg(callback.from_user.id):
        await callback.answer("Нет прав", show_alert=True)
        return
    _, action, oid = callback.data.split(":")
    if action == "pay":
        await callback.message.reply(f"📋 <b>Реквизиты по заказу #{oid}:</b>\n\n{PAYMENT_TEXT}", parse_mode="HTML", reply_markup=get_payment_keyboard())
        await callback.answer()
        return

    status_map = {
        "paid": ("Оплачено (в сборке)", False, f"🎉 Оплата по заказу <b>#{oid}</b> подтверждена!"),
        "ship": ("Отправлено", False, f"📦 Ваш заказ <b>#{oid}</b> отправлен! После получения нажмите «Доставлено» в разделе «Мои заказы»."),
        "canc": ("Отменен", True, f"❌ Ваш заказ <b>#{oid}</b> был отменен."),
    }
    st_label, archived, notice = status_map.get(action, ("Обновлен", False, ""))

    conn = get_db()
    row = conn.execute("SELECT user_tg_id FROM orders WHERE order_id = ?", (oid,)).fetchone()
    conn.execute("UPDATE orders SET status = ?, is_archived = ? WHERE order_id = ?", (st_label, 1 if archived else 0, oid))
    conn.commit()
    conn.close()

    base_text = "\n".join(l for l in (callback.message.text or "").split("\n") if not l.startswith("📌 СТАТУС:"))
    await callback.message.edit_text(f"{base_text}\n\n📌 СТАТУС: {st_label}", reply_markup=callback.message.reply_markup)

    if row and row["user_tg_id"] and notice:
        try:
            await bot.send_message(row["user_tg_id"], notice, parse_mode="HTML")
        except Exception:
            pass
    await callback.answer(f"Статус: {st_label}")


@dp.message(F.reply_to_message)
async def handle_admin_reply(message: types.Message):
    if not is_admin_tg(message.from_user.id):
        await forward_client_message(message)
        return

    replied = message.reply_to_message
    client_id = None
    if replied.forward_from:
        client_id = replied.forward_from.id
    elif replied.text and "ID клиента:" in replied.text:
        try:
            client_id = int(replied.text.split("ID клиента:")[1].strip().split()[0])
        except Exception:
            pass

    if not client_id:
        await message.reply("⚠️ Сделай Reply на сообщение с <code>ID клиента</code>.", parse_mode="HTML")
        return

    reply_text = message.text or message.caption or "[Медиафайл]"
    conn = get_db()
    row = conn.execute("SELECT client_name FROM messages WHERE client_tg_id = ? ORDER BY id DESC LIMIT 1", (client_id,)).fetchone()
    cname = row["client_name"] if row else f"Клиент #{client_id}"
    conn.execute(
        "INSERT INTO messages (client_tg_id, client_name, sender_role, source, text, created_at) VALUES (?, ?, 'admin', 'tg', ?, ?)",
        (client_id, cname, reply_text, time.time()),
    )
    conn.commit()
    conn.close()

    if client_id > 0:
        try:
            await bot.copy_message(chat_id=client_id, from_chat_id=message.chat.id, message_id=message.message_id)
        except Exception:
            pass
    await message.reply("✅ Ответ отправлен клиенту (отобразится и на сайте, и в боте).")


@dp.message()
async def forward_client_message(message: types.Message):
    if is_admin_tg(message.from_user.id):
        return
    uid = message.from_user.id
    cname = message.from_user.full_name
    upsert_user(uid, message.from_user.username or "", cname)

    text_val = message.text or message.caption or "[Вложение из Telegram]"
    conn = get_db()
    conn.execute(
        "INSERT INTO messages (client_tg_id, client_name, sender_role, source, text, created_at) VALUES (?, ?, 'client', 'tg', ?, ?)",
        (uid, cname, text_val, time.time()),
    )
    conn.commit()
    conn.close()

    for adm in get_admin_ids():
        try:
            await bot.send_message(
                adm,
                f"📩 <b>Сообщение из Telegram от {html.escape(cname)}</b>\nID клиента: <code>{uid}</code>",
                parse_mode="HTML",
            )
            await message.forward(adm)
        except Exception:
            pass
    await message.answer("✅ Сообщение передано менеджеру! Ответ появится здесь и в чате на сайте.")
