    function showToast(text, isError = false) {
      const el = document.getElementById("toast");
      el.textContent = text;
      el.classList.toggle("error", isError);
      el.classList.add("show");
      setTimeout(() => el.classList.remove("show"), 2600);
    }

    function formatPrice(num) {
      return Number(num || 0).toLocaleString("ru-RU") + " ₽";
    }

    function getClientChatId() {
      if (tgUser && tgUser.tg_id) return Number(tgUser.tg_id);
      let guestId = localStorage.getItem("atelier_guest_chat_id");
      if (!guestId) {
        guestId = String(-Math.floor(100000 + Math.random() * 899999));
        localStorage.setItem("atelier_guest_chat_id", guestId);
      }
      return Number(guestId);
    }

    async function loadBootstrap() {
      try {
        const res = await fetch("/api/bootstrap");
        const data = await res.json();
        appState = data;
        document.getElementById("patchVersionFooter").textContent = data.version || "1.1";
        data.products.forEach(p => {
          if (!selectedSizes[p.id]) {
            const avail = Object.keys(p.stock || {}).find(sz => Number(p.stock[sz]) > 0);
            selectedSizes[p.id] = avail || Object.keys(p.stock || {})[0] || "M";
          }
        });
        renderMainGrid();
      } catch (e) {
        showToast("Ошибка связи с сервером", true);
      }
    }

    async function initTelegramAuth() {
      if (tgApp && tgApp.initData) {
        try {
          const res = await fetch("/api/auth/telegram", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ initData: tgApp.initData }),
          });
          if (res.ok) {
            const data = await res.json();
            tgUser = data.user;
            localStorage.setItem("atelier_tg_user", JSON.stringify(tgUser));
          }
        } catch (e) {}
      }
      updateAuthButtonUI();
    }

    async function checkSavedAdminToken() {
      if (!adminToken) return;
      try {
        const res = await fetch("/api/admin/verify", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ token: adminToken }),
        });
        if (!res.ok) {
          adminToken = "";
          localStorage.removeItem("atelier_admin_secret");
        }
      } catch (e) {
        adminToken = "";
      }
      updateAdminUI();
    }

    async function verifyAndLoginAdmin() {
      const val = (document.getElementById("adminTokenInput").value || "").trim();
      if (!val) return showToast("Введите пароль", true);

      const res = await fetch("/api/admin/verify", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ token: val }),
      });
      if (!res.ok) {
        showToast("❌ Неверный пароль администратора!", true);
        return;
      }
      adminToken = val;
      localStorage.setItem("atelier_admin_secret", val);
      closeAllModals();
      updateAdminUI();
      showToast("✅ Вход выполнен!");
    }

    function renderMainGrid() {
      const grid = document.getElementById("mainGrid");

      if (catalogMode === "groups" && !activeGroupSlug) {
        grid.innerHTML = appState.groups.map(g => {
          const count = appState.products.filter(p => p.group_slug === g.slug).length;
          return `
            <article class="card" onclick="openGroupProducts('${g.slug}')" style="cursor:pointer;">
              <div class="card-img-wrap">
                <span class="badge-fabric">Группа • ${count} шт.</span>
                ${g.cover ? `<img src="${g.cover}" alt="${g.title}" />` : `<div class="no-photo-box">${g.title.toUpperCase()}</div>`}
              </div>
              <div class="card-body">
                <h3 class="card-title">📁 ${g.title}</h3>
                <p class="card-desc">${g.description || "Нажмите, чтобы посмотреть все вещи этой группы"}</p>
                <button class="add-btn">Открыть подборку →</button>
              </div>
            </article>
          `;
        }).join("");
        return;
      }

      const items = activeGroupSlug
        ? appState.products.filter(p => p.group_slug === activeGroupSlug)
        : appState.products;

      const backHeader = activeGroupSlug
        ? `<div style="grid-column: 1 / -1;"><button class="icon-btn" onclick="switchCatalogMode('groups')">← Назад ко всем группам</button></div>`
        : "";

      if (!items.length) {
        grid.innerHTML = backHeader + `<p style="color:var(--muted); padding:20px;">В этой группе пока нет товаров.</p>`;
        return;
      }

      grid.innerHTML = backHeader + items.map(p => {
        const photo = (p.photos && p.photos[0]) || "";
        const stockEntries = Object.entries(p.stock || {});
        const ratingBadge = p.review_count > 0 ? `★ ${p.rating_avg} (${p.review_count})` : "★ Новый";

        return `
          <article class="card">
            <div class="card-img-wrap" onclick="openProductDetail(${p.id})">
              ${p.fabric ? `<span class="badge-fabric">${p.fabric}</span>` : ""}
              <span class="badge-rating">${ratingBadge}</span>
              ${photo ? `<img src="${photo}" alt="${p.name}" />` : `<div class="no-photo-box">ATELIER</div>`}
            </div>
            <div class="card-body">
              <h3 class="card-title" onclick="openProductDetail(${p.id})">${p.name}</h3>
              <p class="card-desc">${p.description || ""}</p>
              <div class="card-price">${formatPrice(p.price)}</div>

              <div class="sizes">
                ${stockEntries.map(([sz, qty]) => {
                  const isOut = Number(qty) <= 0;
                  const isSel = selectedSizes[p.id] === sz;
                  return `<button class="size-btn ${isSel ? "selected" : ""} ${isOut ? "out" : ""}"
                    onclick="selectCardSize(${p.id}, '${sz}', ${qty})">${sz} <small>(${qty})</small></button>`;
                }).join("")}
              </div>

              <button class="add-btn" onclick="addProductToCart(${p.id})">В корзину</button>

              <div class="card-secondary-btns">
                <button class="card-sec-btn" onclick="openProductDetail(${p.id}, 'sizes')">📏 Сетка и наличие</button>
                <button class="card-sec-btn" onclick="openProductDetail(${p.id}, 'reviews')">⭐ Отзывы (${p.review_count})</button>
              </div>

              ${adminToken ? `
                <div class="card-admin-actions">
                  <button onclick="adminDeleteProduct(${p.id})">🗑 Удалить</button>
                </div>
              ` : ""}
            </div>
          </article>
        `;
      }).join("");
    }

    async function submitProductReview(pid) {
      if (!tgUser) return;
      const rating = Number(document.getElementById("revRatingSelect").value);
      const text = document.getElementById("revTextInput").value.trim();
      const fileInput = document.getElementById("revPhotoInput");
      let photoUrl = "";

      if (fileInput.files && fileInput.files[0]) {
        const fd = new FormData();
        fd.append("files", fileInput.files[0]);
        const upRes = await fetch("/api/upload", { method: "POST", body: fd });
        const upData = await upRes.json();
        photoUrl = (upData.urls && upData.urls[0]) || "";
      }

      const res = await fetch(`/api/products/${pid}/reviews`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ tg_id: tgUser.tg_id, rating, text, photo_url: photoUrl }),
      });

      if (res.ok) {
        showToast("✅ Отзыв опубликован!");
        await loadBootstrap();
        openProductDetail(pid);
      } else {
        const err = await res.json();
        showToast(err.detail || "Ошибка публикации", true);
      }
    }

    function renderOrdersTab(tab) {
      document.getElementById("ordTabActive").classList.toggle("active", tab === "active");
      document.getElementById("ordTabArchive").classList.toggle("active", tab === "archive");
      const list = ordersCache[tab] || [];
      const body = document.getElementById("ordersBody");

      if (!list.length) {
        body.innerHTML = `<p style="font-size:13px; color:var(--muted); text-align:center; padding:20px;">Список пуст</p>`;
        return;
      }

      body.innerHTML = list.map(o => `
        <div class="order-box">
          <div style="display:flex; justify-content:space-between; margin-bottom:6px;">
            <b>Заказ #${o.order_id}</b>
            <span style="color:var(--success); font-weight:700;">${o.status}</span>
          </div>
          <div style="font-size:12px; color:var(--muted); margin-bottom:6px;">
            ${o.items.map(i => `${i.name} (${i.size}) × ${i.qty}`).join("<br/>")}
          </div>
          <div>Итого: <b>${formatPrice(o.total_amount)}</b> • Адрес: ${o.address}</div>
          ${tab === "active" ? `
            <button class="primary-btn green" style="margin-top:10px; padding:9px;" onclick="confirmOrderDelivered('${o.order_id}')">
              ✅ Подтвердить получение (Доставлено)
            </button>
          ` : ""}

      `).join("");
    }

    async function refreshClientChat() {
      const cid = getClientChatId();
      const res = await fetch(`/api/chat/${cid}`);
      const data = await res.json();
      const box = document.getElementById("clientChatBox");
      if (!data.messages.length) {
        box.innerHTML = `<p style="font-size:12px; color:var(--muted); text-align:center; margin-top:40px;">Напишите нам — сообщение сразу уйдет менеджеру в Telegram и на сайт.</p>`;
        return;
      }
      box.innerHTML = data.messages.map(m => `
        <div class="msg-bubble ${m.sender_role}">
          <div>${m.text}</div>
          <div class="msg-meta">${m.sender_role === "admin" ? "Менеджер" : "Вы"}</div>
        </div>
      `).join("");
      box.scrollTop = box.scrollHeight;
    }

    async function sendClientChat() {
      const input = document.getElementById("clientChatInput");
      const text = input.value.trim();
      if (!text) return;
      input.value = "";
      const cid = getClientChatId();
      const cname = tgUser ? tgUser.full_name : `Гость сайта #${Math.abs(cid)}`;

      await fetch("/api/chat/send", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ client_tg_id: cid, client_name: cname, text, source: "web" }),
      });
      await refreshClientChat();
    }

    function addSizeBuilderRow(sz = "", qty = 5, chest = "", shoulders = "", length = "") {
      const row = document.createElement("div");
      row.className = "chart-row-edit";
      row.innerHTML = `
        <input placeholder="Размер" value="${sz}" />

        <input placeholder="Грудь" value="${chest}" />

    async function submitNewProduct() {
      const group_slug = document.getElementById("prodGroupSelect").value;
      const name = document.getElementById("prodNameInput").value.trim();
      const price = Number(document.getElementById("prodPriceInput").value);
      const fabric = document.getElementById("prodFabricInput").value.trim();
      const description = document.getElementById("prodDescInput").value.trim();
      const files = document.getElementById("prodPhotosInput").files;

      if (!name || !price) return showToast("Укажите название и цену", true);

      let photos = [];
      if (files && files.length) {
        const fd = new FormData();
        for (const f of files) fd.append("files", f);
        const up = await (await fetch("/api/upload", { method: "POST", body: fd })).json();
        photos = up.urls || [];
      }

      const stock = {};
      const size_chart = [];
      document.querySelectorAll("#sizeChartBuilder .chart-row-edit").forEach(row => {
        const inputs = row.querySelectorAll("input");
        const sz = inputs[0].value.trim();
        if (sz) {
          stock[sz] = Number(inputs[1].value || 0);
          size_chart.push({
            size: sz,
            chest: inputs[2].value.trim(),
            shoulders: inputs[3].value.trim(),
            length: inputs[4].value.trim(),
          });
        }
      });

      const res = await fetch("/api/products", {
        method: "POST",
        headers: { "Content-Type": "application/json", "X-Admin-Token": adminToken },
        body: JSON.stringify({ group_slug, name, price, fabric, description, photos, stock, size_chart }),
      });

      if (res.ok) {
        closeAllModals();
        showToast("✅ Товар опубликован!");
        await loadBootstrap();
      }
    }

    async function adminDeleteProduct(pid) {
      if (!confirm("Удалить товар?")) return;
      await fetch(`/api/products/${pid}`, {
        method: "DELETE",
        headers: { "X-Admin-Token": adminToken },
      });
      showToast("🗑 Товар удален");
      await loadBootstrap();
    }

    function updateCartBadge() {
      document.getElementById("cartCount").textContent = cart.reduce((s, i) => s + i.qty, 0);
      localStorage.setItem("atelier_cart_v2", JSON.stringify(cart));
    }

    function addProductToCart(pid) {
      const p = appState.products.find(x => x.id === pid);
      const sz = selectedSizes[pid];
      const avail = Number((p.stock || {})[sz] || 0);
      if (avail <= 0) return showToast("Выбранный размер закончился", true);

      const ex = cart.find(i => i.product_id === pid && i.size === sz);
      if (ex) ex.qty += 1;
      else cart.push({ product_id: pid, name: p.name, price: p.price, size: sz, qty: 1 });
      updateCartBadge();
      showToast(`Добавлено: ${p.name} (${sz})`);
    }

    function openCartModal() {
      const body = document.getElementById("cartBody");
      if (!cart.length) {
        body.innerHTML = `<p style="color:var(--muted); text-align:center; padding:20px;">Корзина пуста</p>`;
        return openModal("cartModal");
      }
      const sub = cart.reduce((s, i) => s + i.price * i.qty, 0);
      const deliv = sub >= 5000 ? 0 : 350;
      const total = sub + deliv;
      const defaultName = tgUser ? tgUser.full_name : "";

      body.innerHTML = `
        ${cart.map((i, idx) => `
          <div class="order-box" style="display:flex; justify-content:space-between; align-items:center;">
            <div><b>${i.name}</b><br/><small>Размер: ${i.size} • ${formatPrice(i.price)}</small></div>
            <div>
              <button class="btn-outline" onclick="changeCartQty(${idx}, -1)">−</button>
              <b style="margin:0 6px;">${i.qty}</b>
              <button class="btn-outline" onclick="changeCartQty(${idx}, 1)">+</button>
            </div>
          </div>
        `).join("")}
        <div class="form-group"><label>Имя и телефон</label><input class="input" id="ordName" value="${defaultName}" placeholder="Имя, +7..." /></div>
        <div class="form-group"><label>Адрес доставки (СДЭК / Почта)</label><input class="input" id="ordAddr" placeholder="Город, адрес ПВЗ" /></div>
        <div style="margin-top:12px; font-weight:800;">Итого (с доставкой ${deliv ? deliv + " ₽" : "бесплатно"}): ${formatPrice(total)}</div>
        <button class="primary-btn" onclick="checkoutCart(${deliv}, ${total})">Оформить заказ →</button>
      `;
      openModal("cartModal");
    }

    function changeCartQty(idx, delta) {
      cart[idx].qty += delta;
      if (cart[idx].qty <= 0) cart.splice(idx, 1);
      updateCartBadge();
      openCartModal();
    }

    async function checkoutCart(delivery_cost, total_amount) {
      const customer_name = document.getElementById("ordName").value.trim();
      const address = document.getElementById("ordAddr").value.trim();
      if (!customer_name || !address) return showToast("Заполните имя и адрес", true);

      const res = await fetch("/api/orders", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          user_tg_id: tgUser ? tgUser.tg_id : 0,
          customer_name,
          address,
          items: cart,
          delivery_cost,
          total_amount,
        }),
      });
      const data = await res.json();
      if (res.ok) {
        cart = [];
        updateCartBadge();
        closeAllModals();
        showToast(`✅ Заказ #${data.order_id} оформлен!`);
        await loadBootstrap();
      }
    }
